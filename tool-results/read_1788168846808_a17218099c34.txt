    82→    82→function Calendar({ value, onChange }: { value: string | null; onChange: (iso: string) => void }) {
    83→    83→  const today = new Date();
    84→    84→  today.setHours(0, 0, 0, 0);
    85→    85→  const [view, setView] = useState(() => {
    86→    86→    const base = value ? parseISODate(value) : new Date();
    87→    87→    return new Date(base.getFullYear(), base.getMonth(), 1);
    88→    88→  });
    89→    89→
    90→    90→  const year = view.getFullYear();
    91→    91→  const month = view.getMonth();
    92→    92→  const firstDow = new Date(year, month, 1).getDay();
    93→    93→  const daysInMonth = new Date(year, month + 1, 0).getDate();
    94→    94→  const cells: (string | null)[] = [
    95→    95→    ...Array.from({ length: firstDow }, () => null),
    96→    96→    ...Array.from({ length: daysInMonth }, (_, i) => toISODate(new Date(year, month, i + 1))),
    97→    97→  ];
    98→    98→  const canGoPrev = year > today.getFullYear() || (year === today.getFullYear() && month > today.getMonth());
    99→    99→
   100→   100→  return (
   101→   101→    <div className="rounded-[14px] border border-white/[0.08] bg-panel p-4">
   102→   102→      <div className="mb-5 flex items-center justify-between">
   103→   103→        <button
   104→   104→          type="button"
   105→   105→          aria-label="Previous month"
   106→   106→          disabled={!canGoPrev}
   107→   107→          onClick={() => setView(new Date(year, month - 1, 1))}
   108→   108→          className="text-gold transition hover:text-gold-light disabled:opacity-25"
   109→   109→        >
   110→   110→          <ChevronLeft className="h-6 w-6" />
   111→   111→        </button>
   112→   112→        <span className="font-display text-lg uppercase text-white" aria-live="polite">
   113→   113→          {view.toLocaleDateString("en-US", { month: "long", year: "numeric" })}
   114→   114→        </span>
   115→   115→        <button
   116→   116→          type="button"
   117→   117→          aria-label="Next month"
   118→   118→          onClick={() => setView(new Date(year, month + 1, 1))}
   119→   119→          className="text-gold transition hover:text-gold-light"
   120→   120→        >
   121→   121→          <ChevronRight className="h-6 w-6" />
   122→   122→        </button>
   123→   123→      </div>
   124→   124→
   125→   125→      <div className="mb-2 grid grid-cols-7 gap-1.5 text-center">
   126→   126→        {WEEKDAYS.map((w) => (
   127→   127→          <span key={w} className="text-[9px] font-extrabold tracking-wider text-gold">
   128→   128→            {w}
   129→   129→          </span>
   130→   130→        ))}
   131→   131→      </div>
   132→   132→
   133→   133→      <div className="grid grid-cols-7 gap-1.5" role="grid" aria-label="Reservation calendar">
   134→   134→        {cells.map((iso, i) => {
   135→   135→          if (!iso) return <div key={`empty-${i}`} aria-hidden />;
   136→   136→          const d = parseISODate(iso);
   137→   137→          const past = d.getTime() < today.getTime();
   138→   138→          const thu = d.getDay() === 4;
   139→   139→          const selected = value === iso;
   140→   140→          if (past || !thu) {
   141→   141→            return (
   142→   142→              <div
   143→   143→                key={iso}
   144→   144→                aria-hidden
   145→   145→                className="grid h-9 cursor-not-allowed place-items-center rounded-full text-[11px] text-white/20"
   146→   146→              >
   147→   147→                {d.getDate()}
   148→   148→              </div>
   149→   149→            );
   150→   150→          }
   151→   151→          return (
   152→   152→            <button
   153→   153→              key={iso}
   154→   154→              type="button"
   155→   155→              onClick={() => onChange(iso)}
   156→   156→              aria-pressed={selected}
   157→   157→              aria-label={formatDayLong(iso)}
   158→   158→              className={`grid h-9 place-items-center rounded-full border text-[11px] transition ${
   159→   159→                selected
   160→   160→                  ? "border-gold bg-gold font-extrabold text-[#050505]"
   161→   161→                  : "border-gold/30 text-gold-light hover:bg-gold/10"
   162→   162→              }`}
   163→   163→            >
   164→   164→              {d.getDate()}
   165→   165→            </button>
   166→   166→          );
   167→   167→        })}
   168→   168→      </div>
   169→   169→
   170→   170→      <p className="mt-3 text-center text-[10px] text-white/35">
   171→   171→        Game &amp; Groove runs every Thursday — pick any highlighted date.
   172→   172→      </p>
   173→   173→    </div>
   174→   174→  );
   175→   175→}

   223→   223→
   224→   224→function TableStep({
   225→   225→  draft,
   226→   226→  patch,
   227→   227→  onNext,
   228→   228→}: {
   229→   229→  draft: ReserveDraft;
   230→   230→  patch: (p: Partial<ReserveDraft>) => void;
   231→   231→  onNext: () => void;
   232→   232→}) {
   233→   233→  const [data, setData] = useState<{ key: string; tables: GbTable[] } | null>(null);
   234→   234→  const [refreshing, setRefreshing] = useState(false);
   235→   235→  const [reloadKey, setReloadKey] = useState(0);
   236→   236→  const dataKey = `${draft.date ?? ""}-${draft.slot}`;
   237→   237→  const loading = !data || data.key !== dataKey;
   238→   238→
   239→   239→  useEffect(() => {
   240→   240→    if (!draft.date) return;
   241→   241→    let alive = true;
   242→   242→    const key = `${draft.date}-${draft.slot}`;
   243→   243→    gbApi.tables(draft.date, draft.slot).then(
   244→   244→      (res) => {
   245→   245→        if (!alive) return;
   246→   246→        setData({ key, tables: res.tables });
   247→   247→      },
   248→   248→      (err: unknown) => {
   249→   249→        if (!alive) return;
   250→   250→        toast.error((err as Error).message);
   251→   251→        setData({ key, tables: [] });
   252→   252→      }
   253→   253→    );
   254→   254→    return () => {
   255→   255→      alive = false;
   256→   256→    };
   257→   257→  }, [reloadKey, draft.date, draft.slot]);
   258→   258→
   259→   259→  const tables = loading ? null : (data?.tables ?? []);
   260→   260→  const remaining = tables?.filter((t) => t.available).length ?? 0;
   261→   261→  const selectedTaken = Boolean(
   262→   262→    tables && draft.table && tables.some((t) => t.code === draft.table?.code && !t.available)
   263→   263→  );
   264→   264→
   265→   265→  const refresh = () => {
   266→   266→    setRefreshing(true);
   267→   267→    setReloadKey((k) => k + 1);
   268→   268→    window.setTimeout(() => setRefreshing(false), 900);
   269→   269→  };
   270→   270→
   271→   271→  return (
   272→   272→    <div>
   273→   273→      <div className="mb-4 flex flex-wrap items-center gap-2">
   274→   274→        {draft.date && (
   275→   275→          <span className="inline-flex items-center gap-1.5 rounded-full border border-gold/30 bg-black/50 px-3 py-1.5 text-[11px] font-semibold text-gold-light">
   276→   276→            <MapPin className="h-3 w-3" aria-hidden />
   277→   277→            {formatDayShort(draft.date).dow} {formatDayShort(draft.date).day} · {SLOT_META[draft.slot].short}
   278→   278→          </span>
   279→   279→        )}
   280→   280→        {tables && (
   281→   281→          <span className="text-[11px] text-white/45">
   282→   282→            {remaining} of {tables.length} tables free
   283→   283→          </span>
   284→   284→        )}
   285→   285→      </div>
   286→   286→
   287→   287→      {selectedTaken && (
   288→   288→        <div className="mb-4 rounded-xl border border-amber-500/30 bg-amber-500/10 p-3 text-[11px] leading-relaxed text-amber-300">
   289→   289→          Table {draft.table?.code} was just booked by someone else — please pick another table.
   290→   290→        </div>
   291→   291→      )}
   292→   292→
   293→   293→      {loading ? (
   294→   294→        <GbSkeleton className="h-[310px] w-full" />
   295→   295→      ) : (
   296→   296→        <div className="relative mb-4 h-[310px] overflow-hidden rounded-2xl border border-gold/25 bg-[#0c0c0c]">
   297→   297→          <div
   298→   298→            aria-hidden
   299→   299→            className="pointer-events-none absolute inset-0 bg-[radial-gradient(circle_at_50%_100%,rgba(245,189,25,0.07),transparent_50%)]"
   300→   300→          />
   301→   301→          <div className="absolute left-1/2 top-4 w-[150px] -translate-x-1/2 rounded border border-white/15 py-1.5 text-center text-[9px] tracking-wider text-white/50">
   302→   302→            SCREEN
   303→   303→          </div>
   304→   304→
   305→   305→          {tables?.map((t) => {
   306→   306→            const selected = draft.table?.code === t.code;
   307→   307→            const base =
   308→   308→              "absolute h-[42px] w-[55px] rounded-lg border text-[11px] font-extrabold transition grid place-items-center";
   309→   309→            if (!t.available) {
   310→   310→              return (
   311→   311→                <button
   312→   312→                  key={t.code}
   313→   313→                  type="button"
   314→   314→                  disabled
   315→   315→                  aria-label={`Table ${t.code} is booked`}
   316→   316→                  className={`${base} cursor-not-allowed border-white/[0.06] bg-white/[0.03] text-white/25 ${TABLE_POSITIONS[t.code] ?? ""}`}
   317→   317→                >
   318→   318→                  <span>
   319→   319→                    <X className="mx-auto h-3 w-3 text-red-400/70" aria-hidden />
   320→   320→                    <span className="mt-0.5 block text-[8px] tracking-wider">{t.code}</span>
   321→   321→                  </span>
   322→   322→                </button>
   323→   323→              );
   324→   324→            }
   325→   325→            return (
   326→   326→              <button
   327→   327→                key={t.code}
   328→   328→                type="button"
   329→   329→                onClick={() => patch({ table: t })}
   330→   330→                aria-pressed={selected}
   331→   331→                aria-label={`Table ${t.code}, ${t.zone}, seats ${t.seats}`}
   332→   332→                className={`${base} ${
   333→   333→                  selected
   334→   334→                    ? "border-gold bg-gold text-[#050505]"
   335→   335→                    : "border-gold-dark bg-[#111] text-gold-light hover:border-gold-light"
   336→   336→                } ${TABLE_POSITIONS[t.code] ?? ""}`}
   337→   337→              >
   338→   338→                {t.code}
   339→   339→              </button>
   340→   340→            );
   341→   341→          })}
   342→   342→
   343→   343→          <div className="absolute bottom-5 left-1/2 flex h-[42px] w-[180px] -translate-x-1/2 items-center justify-center gap-1.5 rounded-lg border border-gold font-display text-[11px] font-bold tracking-wider text-gold">

   362→   362→        <span className="flex items-center gap-1.5">
   363→   363→          <span className="h-2.5 w-2.5 rounded bg-gold" aria-hidden /> SELECTED

   431→   431→      </div>
   432→   432→
   433→   433→      <GbInfoCard
   434→   434→        label="PARTY SIZE"
   435→   435→        value={`${draft.guests} ${draft.guests === 1 ? "Guest" : "Guests"}`}
   436→   436→        sub={
   437→   437→          draft.table
   438→   438→            ? `Table ${draft.table.code} fits up to ${max} guests`
   439→   439→            : "Select a table to see its capacity"
   440→   440→        }

   532→   532→        )}

   534→   534→          <span className="inline-flex items-center gap-1.5 rounded-full border border-gold/30 bg-black/50 px-3 py-1.5 text-[11px] font-semibold text-gold-light">
   535→   535→            <Users className="h-3 w-3" aria-hidden />
   536→   536→            {draft.table.code} · {draft.guests} guests
   537→   537→          </span>
   538→   538→        )}
   539→   539→      </div>
   540→   540→
   541→   541→      <div className="mb-4">
   542→   542→        <label htmlFor="gb-name" className="mb-1.5 block text-[10px] tracking-[0.15em] text-white/45">
   543→   543→          FULL NAME
   544→   544→        </label>
   545→   545→        <input
   546→   546→          id="gb-name"
   547→   547→          type="text"
   548→   548→          autoComplete="name"
   549→   549→          value={name}
   550→   550→          onChange={(e) => setName(e.target.value)}
   551→   551→          placeholder="Your name"
   552→   552→          className="w-full rounded-[10px] border border-gold/25 bg-panel px-3.5 py-3.5 text-[13px] text-white outline-none transition placeholder:text-white/25 focus:border-gold"
   553→   553→        />
   554→   554→        {errors.name && <p className="mt-1.5 text-[11px] text-red-400">{errors.name}</p>}
   555→   555→      </div>
   556→   556→
   557→   557→      <div className="mb-4">

   681→   681→
   682→   682→export function GbReserve({
   683→   683→  draft,
   684→   684→  onChange,
   685→   685→  savedDetails,
   686→   686→  onSaveDetails,
   687→   687→  onHome,
   688→   688→  onViewReservations,
   689→   689→}: {
   690→   690→  draft: ReserveDraft;
   691→   691→  onChange: (d: ReserveDraft) => void;
   692→   692→  savedDetails: SavedDetails | null;
   693→   693→  onSaveDetails: (d: SavedDetails | null) => void;
   694→   694→  onHome: () => void;
   695→   695→  onViewReservations: () => void;
   696→   696→}) {
   697→   697→  const [step, setStep] = useState<Step>("date");
   698→   698→  const patch = (p: Partial<ReserveDraft>) => onChange({ ...draft, ...p });
   699→   699→  const stepIndex = STEPS.findIndex((s) => s.key === step);
   700→   700→
   701→   701→  const back = () => {
   702→   702→    if (step === "date") onHome();
   703→   703→    else if (step === "table") setStep("date");
   704→   704→    else if (step === "guests") setStep("table");
   705→   705→    else if (step === "details") setStep("guests");
   706→   706→  };
   707→   707→
   708→   708→  const titles: Record<Step, string> = {
   709→   709→    date: "Choose Date",
   710→   710→    table: "Choose Your Table",
   711→   711→    guests: "How Many Guests?",
   712→   712→    details: "Your Details",
   713→   713→    done: "Confirmed",
   714→   714→  };
   715→   715→
   716→   716→  return (
   717→   717→    <div className="px-5 pb-9 pt-6">
   718→   718→      {step !== "done" && (
   719→   719→        <>
   720→   720→          <div className="mb-6 flex items-center gap-4">
   721→   721→            <button
   722→   722→              type="button"
   723→   723→              aria-label="Go back"
   724→   724→              onClick={back}
   725→   725→              className="grid h-9 w-9 place-items-center rounded-xl border border-white/10 text-white transition hover:border-gold/40 hover:text-gold"
   726→   726→            >
   727→   727→              <ChevronLeft className="h-5 w-5" />
   728→   728→            </button>
   729→   729→            <h1 className="font-display text-2xl uppercase text-white">{titles[step]}</h1>
   730→   730→          </div>
   731→   731→
   732→   732→          <div className="mb-7" aria-hidden>
   733→   733→            <div className="flex gap-1.5">
   734→   734→              {STEPS.map((s, i) => (
   735→   735→                <span
   736→   736→                  key={s.key}
   737→   737→                  className={`h-1.5 flex-1 rounded-full transition-colors ${
   738→   738→                    i <= stepIndex ? "bg-gold" : "bg-white/15"
   739→   739→                  }`}
   740→   740→                />
   741→   741→              ))}
   742→   742→            </div>
   743→   743→            <p className="mt-2 text-[10px] font-bold tracking-[0.2em] text-white/40">
   744→   744→              STEP {stepIndex + 1} OF 4 · {STEPS[stepIndex].label}
   745→   745→            </p>
   746→   746→          </div>
   747→   747→        </>
   748→   748→      )}
   749→   749→
   750→   750→      {step === "date" && <DateStep draft={draft} patch={patch} onNext={() => setStep("table")} />}
   751→   751→
   752→   752→      {step === "table" && <TableStep draft={draft} patch={patch} onNext={() => setStep("guests")} />}
   753→   753→
   754→   754→      {step === "guests" && <GuestsStep draft={draft} patch={patch} onNext={() => setStep("details")} />}
   755→   755→
   756→   756→      {step === "details" && (
   757→   757→        <DetailsStep
   758→   758→          draft={draft}
   759→   759→          savedDetails={savedDetails}
   760→   760→          onSaveDetails={onSaveDetails}
   761→   761→          onTableConflict={() => {
   762→   762→            patch({ table: null });
   763→   763→            setStep("table");
   764→   764→          }}
   765→   765→          onDone={(r) => {
   766→   766→            patch({ reservation: r });
   767→   767→            setStep("done");
   768→   768→          }}
   769→   769→        />
   770→   770→      )}
   771→   771→
   772→   772→      {step === "done" && draft.reservation && (
   773→   773→        <DoneStep reservation={draft.reservation} onHome={onHome} onViewReservations={onViewReservations} />
   774→   774→      )}
   775→   775→
   776→   776→      {step === "done" && !draft.reservation && (
   777→   777→        <div className="pt-8 text-center">
   778→   778→          <p className="text-[13px] text-white/60">Something went wrong — let&apos;s start again.</p>
   779→   779→          <GbSecondaryButton className="mt-4" onClick={onHome}>
   780→   780→            BACK TO HOME
   781→   781→          </GbSecondaryButton>
   782→   782→        </div>
   783→   783→      )}
   784→   784→    </div>
   785→   785→  );
   786→   786→}
