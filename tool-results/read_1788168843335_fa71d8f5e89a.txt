    49→function downloadIcs(r: GbReservation) {
    50→  const blob = new Blob([buildIcs(r)], { type: "text/calendar;charset=utf-8" });
    51→  const url = URL.createObjectURL(blob);
    52→  const a = document.createElement("a");
    53→  a.href = url;
    54→  a.download = `gura-bicha-${r.code}.ics`;
    55→  document.body.appendChild(a);
    56→  a.click();
    57→  a.remove();
    58→  URL.revokeObjectURL(url);
    59→  toast.success("Calendar file downloaded");
    60→}

    62→async function shareReservation(r: GbReservation) {
    63→  const text = `See you at Gura Bicha! ${r.code} · Table ${r.tableCode} · ${formatDayLong(r.date)} · ${SLOT_META[r.slot].time}`;
    64→  if (typeof navigator !== "undefined" && navigator.share) {
    65→    try {
    66→      await navigator.share({ title: "Gura Bicha Sports Bar", text });
    67→      return;
    68→    } catch {
    69→      // user cancelled or share failed — fall through to clipboard
    70→    }
    71→  }
    72→  try {
    73→    await navigator.clipboard.writeText(text);
    74→    toast.success("Invitation copied to clipboard");
    75→  } catch {
    76→    toast.error("Could not share — please copy your reservation code manually.");
    77→  }

    80→/* ============ CALENDAR (Thursdays = event nights) ============ */

    82→function Calendar({ value, onChange }: { value: string | null; onChange: (iso: string) => void }) {
    83→  const today = new Date();
    84→  today.setHours(0, 0, 0, 0);
    85→  const [view, setView] = useState(() => {
    86→    const base = value ? parseISODate(value) : new Date();
    87→    return new Date(base.getFullYear(), base.getMonth(), 1);
    88→  });
    89→
    90→  const year = view.getFullYear();
    91→  const month = view.getMonth();
    92→  const firstDow = new Date(year, month, 1).getDay();
    93→  const daysInMonth = new Date(year, month + 1, 0).getDate();
    94→  const cells: (string | null)[] = [
    95→    ...Array.from({ length: firstDow }, () => null),
    96→    ...Array.from({ length: daysInMonth }, (_, i) => toISODate(new Date(year, month, i + 1))),
    97→  ];
    98→  const canGoPrev = year > today.getFullYear() || (year === today.getFullYear() && month > today.getMonth());
    99→
   100→  return (
   101→    <div className="rounded-[14px] border border-white/[0.08] bg-panel p-4">
   102→      <div className="mb-5 flex items-center justify-between">
   103→        <button
   104→          type="button"
   105→          aria-label="Previous month"
   106→          disabled={!canGoPrev}
   107→          onClick={() => setView(new Date(year, month - 1, 1))}
   108→          className="text-gold transition hover:text-gold-light disabled:opacity-25"
   109→        >
   110→          <ChevronLeft className="h-6 w-6" />
   111→        </button>
   112→        <span className="font-display text-lg uppercase text-white" aria-live="polite">
   113→          {view.toLocaleDateString("en-US", { month: "long", year: "numeric" })}
   114→        </span>
   115→        <button
   116→          type="button"
   117→          aria-label="Next month"
   118→          onClick={() => setView(new Date(year, month + 1, 1))}
   119→          className="text-gold transition hover:text-gold-light"
   120→        >
   121→          <ChevronRight className="h-6 w-6" />
   122→        </button>
   123→      </div>
   124→
   125→      <div className="mb-2 grid grid-cols-7 gap-1.5 text-center">
   126→        {WEEKDAYS.map((w) => (
   127→          <span key={w} className="text-[9px] font-extrabold tracking-wider text-gold">
   128→            {w}
   129→          </span>
   130→        ))}
   131→      </div>
   132→
   133→      <div className="grid grid-cols-7 gap-1.5" role="grid" aria-label="Reservation calendar">
   134→        {cells.map((iso, i) => {
   135→          if (!iso) return <div key={`empty-${i}`} aria-hidden />;
   136→          const d = parseISODate(iso);
   137→          const past = d.getTime() < today.getTime();
   138→          const thu = d.getDay() === 4;
   139→          const selected = value === iso;
   140→          if (past || !thu) {
   141→            return (
   142→              <div
   143→                key={iso}
   144→                aria-hidden
   145→                className="grid h-9 cursor-not-allowed place-items-center rounded-full text-[11px] text-white/20"
   146→              >
   147→                {d.getDate()}
   148→              </div>
   149→            );
   150→          }
   151→          return (
   152→            <button
   153→              key={iso}
   154→              type="button"
   155→              onClick={() => onChange(iso)}
   156→              aria-pressed={selected}
   157→              aria-label={formatDayLong(iso)}
   158→              className={`grid h-9 place-items-center rounded-full border text-[11px] transition ${
   159→                selected
   160→                  ? "border-gold bg-gold font-extrabold text-[#050505]"
   161→                  : "border-gold/30 text-gold-light hover:bg-gold/10"
   162→              }`}
   163→            >
   164→              {d.getDate()}
   165→            </button>
   166→          );
   167→        })}
   168→      </div>
   169→
   170→      <p className="mt-3 text-center text-[10px] text-white/35">
   171→        Game &amp; Groove runs every Thursday — pick any highlighted date.
   172→      </p>
   173→    </div>
   174→  );
   175→}

   222→/* ============ STEP: TABLE ============ */
   223→
   224→function TableStep({
   225→  draft,
   226→  patch,
   227→  onNext,
   228→}: {
   229→  draft: ReserveDraft;
   230→  patch: (p: Partial<ReserveDraft>) => void;
   231→  onNext: () => void;
   232→}) {
   233→  const [data, setData] = useState<{ key: string; tables: GbTable[] } | null>(null);
   234→  const [refreshing, setRefreshing] = useState(false);
   235→  const [reloadKey, setReloadKey] = useState(0);
   236→  const dataKey = `${draft.date ?? ""}-${draft.slot}`;
   237→  const loading = !data || data.key !== dataKey;
   238→
   239→  useEffect(() => {
   240→    if (!draft.date) return;
   241→    let alive = true;
   242→    const key = `${draft.date}-${draft.slot}`;
   243→    gbApi.tables(draft.date, draft.slot).then(
   244→      (res) => {
   245→        if (!alive) return;
   246→        setData({ key, tables: res.tables });
   247→      },
   248→      (err: unknown) => {
   249→        if (!alive) return;
   250→        toast.error((err as Error).message);
   251→        setData({ key, tables: [] });
   252→      }
   253→    );
   254→    return () => {
   255→      alive = false;
   256→    };
   257→  }, [reloadKey, draft.date, draft.slot]);
   258→
   259→  const tables = loading ? null : (data?.tables ?? []);
   260→  const remaining = tables?.filter((t) => t.available).length ?? 0;
   261→  const selectedTaken = Boolean(
   262→    tables && draft.table && tables.some((t) => t.code === draft.table?.code && !t.available)
   263→  );
   264→
   265→  const refresh = () => {
   266→    setRefreshing(true);
   267→    setReloadKey((k) => k + 1);
   268→    window.setTimeout(() => setRefreshing(false), 900);
   269→  };
   270→
   271→  return (
   272→    <div>
   273→      <div className="mb-4 flex flex-wrap items-center gap-2">
   274→        {draft.date && (
   275→          <span className="inline-flex items-center gap-1.5 rounded-full border border-gold/30 bg-black/50 px-3 py-1.5 text-[11px] font-semibold text-gold-light">
   276→            <MapPin className="h-3 w-3" aria-hidden />
   277→            {formatDayShort(draft.date).dow} {formatDayShort(draft.date).day} · {SLOT_META[draft.slot].short}
   278→          </span>
   279→        )}
   280→        {tables && (
   281→          <span className="text-[11px] text-white/45">
   282→            {remaining} of {tables.length} tables free
   283→          </span>
   284→        )}
   285→      </div>
   286→
   287→      {selectedTaken && (
   288→        <div className="mb-4 rounded-xl border border-amber-500/30 bg-amber-500/10 p-3 text-[11px] leading-relaxed text-amber-300">
   289→          Table {draft.table?.code} was just booked by someone else — please pick another table.
   290→        </div>
   291→      )}
   292→
   293→      {loading ? (
   294→        <GbSkeleton className="h-[310px] w-full" />
   295→      ) : (
   296→        <div className="relative mb-4 h-[310px] overflow-hidden rounded-2xl border border-gold/25 bg-[#0c0c0c]">
   297→          <div
   298→            aria-hidden
   299→            className="pointer-events-none absolute inset-0 bg-[radial-gradient(circle_at_50%_100%,rgba(245,189,25,0.07),transparent_50%)]"
   300→          />
   301→          <div className="absolute left-1/2 top-4 w-[150px] -translate-x-1/2 rounded border border-white/15 py-1.5 text-center text-[9px] tracking-wider text-white/50">
   302→            SCREEN
   303→          </div>
   304→
   305→          {tables?.map((t) => {
   306→            const selected = draft.table?.code === t.code;
   307→            const base =
   308→              "absolute h-[42px] w-[55px] rounded-lg border text-[11px] font-extrabold transition grid place-items-center";
   309→            if (!t.available) {
   310→              return (
   311→                <button
   312→                  key={t.code}
   313→                  type="button"
   314→                  disabled
   315→                  aria-label={`Table ${t.code} is booked`}
   316→                  className={`${base} cursor-not-allowed border-white/[0.06] bg-white/[0.03] text-white/25 ${TABLE_POSITIONS[t.code] ?? ""}`}
   317→                >
   318→                  <span>
   319→                    <X className="mx-auto h-3 w-3 text-red-400/70" aria-hidden />
   320→                    <span className="mt-0.5 block text-[8px] tracking-wider">{t.code}</span>
   321→                  </span>
   322→                </button>
   323→              );
   324→            }
   325→            return (
   326→              <button
   327→                key={t.code}
   328→                type="button"
   329→                onClick={() => patch({ table: t })}
   330→                aria-pressed={selected}
   331→                aria-label={`Table ${t.code}, ${t.zone}, seats ${t.seats}`}
   332→                className={`${base} ${
   333→                  selected
   334→                    ? "border-gold bg-gold text-[#050505]"
   335→                    : "border-gold-dark bg-[#111] text-gold-light hover:border-gold-light"
   336→                } ${TABLE_POSITIONS[t.code] ?? ""}`}
   337→              >
   338→                {t.code}
   339→              </button>
   340→            );
   341→          })}
   342→
   343→          <div className="absolute bottom-5 left-1/2 flex h-[42px] w-[180px] -translate-x-1/2 items-center justify-center gap-1.5 rounded-lg border border-gold font-display text-[11px] font-bold tracking-wider text-gold">
   344→            STAGE
   345→          </div>
   346→
   347→          <button
   348→            type="button"
   349→            aria-label="Refresh availability"
   350→            onClick={refresh}
   351→            className="absolute right-3 top-3 grid h-8 w-8 place-items-center rounded-lg border border-white/10 text-white/50 transition hover:text-gold"
   352→          >
   353→            <RotateCw className={`h-3.5 w-3.5 ${refreshing ? "animate-spin" : ""}`} />
   354→          </button>
   355→        </div>
   356→      )}
   357→
   358→      <div className="mb-4 flex items-center justify-center gap-4 text-[9px] font-bold tracking-wider text-white/40">
   359→        <span className="flex items-center gap-1.5">
   360→          <span className="h-2.5 w-2.5 rounded border border-gold-dark bg-[#111]" aria-hidden /> AVAILABLE
   361→        </span>
   362→        <span className="flex items-center gap-1.5">
   363→          <span className="h-2.5 w-2.5 rounded bg-gold" aria-hidden /> SELECTED
   364→        </span>
   365→        <span className="flex items-center gap-1.5">
   366→          <span className="h-2.5 w-2.5 rounded bg-white/10" aria-hidden /> BOOKED
   367→        </span>
   368→      </div>
   369→
   370→      <GbInfoCard
   371→        label="SELECTED TABLE"
   372→        value={draft.table ? draft.table.code : "Tap a table on the map"}
   373→        sub={draft.table ? `${draft.table.zone} · ${draft.table.seats}` : "Gold tables are still open"}
   374→      />
   375→
   376→      <GbPrimaryButton disabled={!draft.table || selectedTaken} onClick={onNext}>
   377→        CONTINUE
   378→      </GbPrimaryButton>
   379→    </div>
   380→  );
   381→}

   448→/* ============ STEP: DETAILS ============ */
   449→
   450→function DetailsStep({
   451→  draft,
   452→  savedDetails,
   453→  onSaveDetails,
   454→  onTableConflict,
   455→  onDone,
   456→}: {
   457→  draft: ReserveDraft;
   458→  savedDetails: SavedDetails | null;
   459→  onSaveDetails: (d: SavedDetails | null) => void;
   460→  onTableConflict: () => void;
   461→  onDone: (r: GbReservation) => void;
   462→}) {
   463→  const [name, setName] = useState(savedDetails?.name ?? "");
   464→  const [phone, setPhone] = useState(savedDetails?.phone ?? "");
   465→  const [email, setEmail] = useState(savedDetails?.email ?? "");
   466→  const [request, setRequest] = useState("");
   467→  const [save, setSave] = useState(Boolean(savedDetails));
   468→  const [errors, setErrors] = useState<{ name?: string; phone?: string }>({});
   469→  const [submitting, setSubmitting] = useState(false);
   470→
   471→  const submit = async () => {
   472→    const nextErrors: { name?: string; phone?: string } = {};
   473→    if (name.trim().length < 2) nextErrors.name = "Please enter your full name.";
   474→    if (phone.replace(/\D/g, "").length < 7) nextErrors.phone = "Enter a valid phone number (e.g. +251 911 234 567).";
   475→    setErrors(nextErrors);
   476→    if (Object.keys(nextErrors).length > 0 || !draft.date || !draft.table) return;
   477→
   478→    setSubmitting(true);
   479→    try {
   480→      const res = await gbApi.createReservation({
   481→        date: draft.date,
   482→        slot: draft.slot,
   483→        tableCode: draft.table.code,
   484→        guests: draft.guests,
   485→        name: name.trim(),
   486→        phone: phone.trim(),
   487→        email: email.trim() || undefined,
   488→        request: request.trim() || undefined,
   489→      });
   490→      if (save) {
   491→        const details: SavedDetails = { name: name.trim(), phone: phone.trim(), email: email.trim() };
   492→        onSaveDetails(details);
   493→        try {
   494→          localStorage.setItem("gb:details", JSON.stringify(details));
   495→        } catch {
   496→          // storage unavailable — ignore
   497→        }
   498→      } else {
   499→        onSaveDetails(null);
   500→        try {
   501→          localStorage.removeItem("gb:details");
   502→        } catch {
   503→          // ignore
   504→        }
   505→      }
   506→      toast.success(`Reservation ${res.reservation.code} confirmed`);
   507→      onDone(res.reservation);
   508→    } catch (err) {
   509→      const message = (err as Error).message;
   510→      toast.error(message);
   511→      if (message.toLowerCase().includes("no longer available")) {
   512→        onTableConflict();
   513→      }
   514→    } finally {
   515→      setSubmitting(false);
   516→    }
   517→  };
   518→
   519→  return (
   520→    <form
   521→      onSubmit={(e) => {
   522→        e.preventDefault();
   523→        void submit();
   524→      }}
   525→    >
   526→      <div className="mb-5 flex flex-wrap gap-2">
   527→        {draft.date && (
   528→          <span className="inline-flex items-center gap-1.5 rounded-full border border-gold/30 bg-black/50 px-3 py-1.5 text-[11px] font-semibold text-gold-light">
   529→            <Clock className="h-3 w-3" aria-hidden />
   530→            {formatDayLong(draft.date).replace(/,/, " ·")} · {SLOT_META[draft.slot].short}
   531→          </span>
   532→        )}
   533→        {draft.table && (
   534→          <span className="inline-flex items-center gap-1.5 rounded-full border border-gold/30 bg-black/50 px-3 py-1.5 text-[11px] font-semibold text-gold-light">

   680→/* ============ FLOW CONTAINER ============ */
   681→
   682→export function GbReserve({
   683→  draft,
   684→  onChange,
   685→  savedDetails,
   686→  onSaveDetails,
   687→  onHome,
   688→  onViewReservations,
   689→}: {
   690→  draft: ReserveDraft;
   691→  onChange: (d: ReserveDraft) => void;
   692→  savedDetails: SavedDetails | null;
   693→  onSaveDetails: (d: SavedDetails | null) => void;
   694→  onHome: () => void;
   695→  onViewReservations: () => void;
   696→}) {
   697→  const [step, setStep] = useState<Step>("date");
   698→  const patch = (p: Partial<ReserveDraft>) => onChange({ ...draft, ...p });
   699→  const stepIndex = STEPS.findIndex((s) => s.key === step);
   700→
   701→  const back = () => {
   702→    if (step === "date") onHome();
   703→    else if (step === "table") setStep("date");
   704→    else if (step === "guests") setStep("table");
   705→    else if (step === "details") setStep("guests");
   706→  };
   707→
   708→  const titles: Record<Step, string> = {
   709→    date: "Choose Date",
   710→    table: "Choose Your Table",
   711→    guests: "How Many Guests?",
   712→    details: "Your Details",
   713→    done: "Confirmed",
   714→  };
   715→
   716→  return (
   717→    <div className="px-5 pb-9 pt-6">
   718→      {step !== "done" && (
   719→        <>
   720→          <div className="mb-6 flex items-center gap-4">
   721→            <button
   722→              type="button"
   723→              aria-label="Go back"
   724→              onClick={back}
   725→              className="grid h-9 w-9 place-items-center rounded-xl border border-white/10 text-white transition hover:border-gold/40 hover:text-gold"
   726→            >
   727→              <ChevronLeft className="h-5 w-5" />
   728→            </button>
   729→            <h1 className="font-display text-2xl uppercase text-white">{titles[step]}</h1>
   730→          </div>
   731→
   732→          <div className="mb-7" aria-hidden>
   733→            <div className="flex gap-1.5">
   734→              {STEPS.map((s, i) => (
   735→                <span
   736→                  key={s.key}
   737→                  className={`h-1.5 flex-1 rounded-full transition-colors ${
   738→                    i <= stepIndex ? "bg-gold" : "bg-white/15"
   739→                  }`}
   740→                />
   741→              ))}
   742→            </div>
   743→            <p className="mt-2 text-[10px] font-bold tracking-[0.2em] text-white/40">
   744→              STEP {stepIndex + 1} OF 4 · {STEPS[stepIndex].label}
   745→            </p>
   746→          </div>
   747→        </>
   748→      )}
   749→
   750→      {step === "date" && <DateStep draft={draft} patch={patch} onNext={() => setStep("table")} />}
   751→
   752→      {step === "table" && <TableStep draft={draft} patch={patch} onNext={() => setStep("guests")} />}
   753→
   754→      {step === "guests" && <GuestsStep draft={draft} patch={patch} onNext={() => setStep("details")} />}
   755→
   756→      {step === "details" && (
   757→        <DetailsStep
   758→          draft={draft}
   759→          savedDetails={savedDetails}
   760→          onSaveDetails={onSaveDetails}
   761→          onTableConflict={() => {
   762→            patch({ table: null });
   763→            setStep("table");
   764→          }}
   765→          onDone={(r) => {
   766→            patch({ reservation: r });
   767→            setStep("done");
   768→          }}
   769→        />
   770→      )}
   771→
   772→      {step === "done" && draft.reservation && (
   773→        <DoneStep reservation={draft.reservation} onHome={onHome} onViewReservations={onViewReservations} />
   774→      )}
   775→
   776→      {step === "done" && !draft.reservation && (
   777→        <div className="pt-8 text-center">
   778→          <p className="text-[13px] text-white/60">Something went wrong — let&apos;s start again.</p>
   779→          <GbSecondaryButton className="mt-4" onClick={onHome}>
   780→            BACK TO HOME
   781→          </GbSecondaryButton>
   782→        </div>
   783→      )}
   784→    </div>
   785→  );
   786→}
